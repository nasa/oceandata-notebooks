## 3. Understanding the SPEXONE L2 Product Structure

+++

The SPEXONE FastMAPOL L2 `MAPOL_ocean` product includes aerosol optical properties for both fine and coarse aerosol modes, as well as retrieved ocean surface properties.

ocean surface product variables include:

* ocean surface reflectance (`rhos*`)
* BRDF parameters based on the Ross–Li model
* ocean NDVI
* ocean white-sky albedo

Multi-angle ocean surface reflectance is derived by applying atmospheric correction independently to each SPEXONE viewing angle. `rhos_angular` represents the retrieved surface reflectance before BRDF correction, while `rhos_nadir` represents the reflectance after BRDF correction and adjustment to nadir viewing geometry. Angular means and standard deviations are also provided as `rhos_nadir_mean`, `rhos_nadir_std`, `rhos_angular_mean`, and `rhos_angular_std`.

The following lists variables related to the ocean surface model (names beginning with `ocean`) and ocean surface reflectance (names beginning with `rhos`).

### Define helper functions for visualization

+++

The following helper functions are used to visualize single-band and RGB maps, examine the angular dependence of surface reflectance, and calculate reflectance ratios between selected viewing angles.

```{code-cell} ipython3
import matplotlib.pyplot as plt
from matplotlib.colors import LogNorm
import cartopy.crs as ccrs
import cartopy.feature as cfeature
import numpy as np
import math

def _prepare_rgba(data, vmin=None, vmax=None):
    """Normalize RGB/RGBA data and make invalid pixels transparent."""
    data = np.asarray(data, float)

    rgb = data[..., :3].copy()
    alpha = data[..., 3].copy() if data.shape[-1] == 4 else np.ones(data.shape[:2])

    valid = np.all(np.isfinite(rgb), axis=-1) & np.isfinite(alpha)

    if vmin is not None or vmax is not None:
        vmin1 = 0 if vmin is None else vmin
        vmax1 = np.nanmax(rgb) if vmax is None else vmax
        rgb = (rgb - vmin1) / (vmax1 - vmin1)

    rgb = np.clip(np.nan_to_num(rgb), 0, 1)
    alpha = np.where(valid, alpha, 0)

    return np.dstack([rgb, np.clip(alpha, 0, 1)])


def plot_l2_product(
    lon, lat, data, plot_range, label="", title="",
    vmin=None, vmax=None, figsize=(12, 4),
    cmap="viridis", log_scale=False,
    ocean_color="#f2efe9", land_color="#dbe9f6",
):
    """Plot scalar map + histogram, or RGB/RGBA map."""

    data = np.asarray(data)

    # Remove single-band dimension
    if data.ndim == 3 and data.shape[-1] == 1:
        data = data[..., 0]

    is_rgb = data.ndim == 3 and data.shape[-1] in (3, 4)

    # ---------------- Figure ----------------
    if is_rgb:
        fig, ax = plt.subplots(
            figsize=(figsize[0] * 0.75, figsize[1]),
            subplot_kw={"projection": ccrs.PlateCarree()},
        )

    else:
        if data.ndim != 2:
            raise ValueError("data must be 2D scalar or 3/4-band RGB.")

        values = data[np.isfinite(data)]
        if log_scale:
            values = values[values > 0]

        if values.size == 0:
            raise ValueError("No valid data.")

        vmin = np.percentile(values, 2) if vmin is None else vmin
        vmax = np.percentile(values, 98) if vmax is None else vmax

        fig = plt.figure(figsize=figsize)
        gs = fig.add_gridspec(1, 2, width_ratios=[3, 1], wspace=0.3)
        ax = fig.add_subplot(gs[0], projection=ccrs.PlateCarree())

    # ---------------- Background ----------------
    ax.set_extent(plot_range, crs=ccrs.PlateCarree())
    ax.add_feature(cfeature.OCEAN, facecolor=ocean_color, zorder=0)
    ax.add_feature(cfeature.LAND, facecolor=land_color, zorder=0)

    # ---------------- RGB ----------------
    if is_rgb:
        ax.pcolormesh(
            lon, lat, data,
            shading="nearest",
            transform=ccrs.PlateCarree(),
            zorder=1,
        )

    # ---------------- Scalar ----------------
    else:
        norm = LogNorm(vmin=vmin, vmax=vmax) if log_scale else None

        pm = ax.pcolormesh(
            lon, lat, data,
            cmap=cmap,
            norm=norm,
            vmin=None if log_scale else vmin,
            vmax=None if log_scale else vmax,
            shading="auto",
            transform=ccrs.PlateCarree(),
            zorder=1,
        )

        plt.colorbar(pm, ax=ax, pad=0.1).set_label(label)

        # Histogram
        axh = fig.add_subplot(gs[1])

        bins = (
            np.logspace(np.log10(vmin), np.log10(vmax), 40)
            if log_scale else 40
        )

        axh.hist(
            values,
            bins=bins,
            range=None if log_scale else (vmin, vmax),
            color="gray",
            edgecolor="black",
        )

        if log_scale:
            axh.set_xscale("log")

        axh.set(
            xlabel=label,
            ylabel="Count",
            title=f"Histogram: N={values.size}",
        )

    # ---------------- Decoration ----------------
    ax.coastlines(resolution="110m", linewidth=0.8, zorder=3)
    ax.gridlines(draw_labels=True, zorder=4)
    ax.set_title(title)

    plt.show()

def plot_spectrum(
    dataset, var, iy, ix, n=1,
    wavelength_var="wavelength",
    xlim=None,
    ylim=None,
    title=None,
    ylabel=None,
    figsize=(7, 4),
    print_stats=False,
    ax=None,
):
    """
    Plot spectral variation of any variable with dimensions
    (lat, lon, wavelength).

    Examples: Rrs, AOD, SSA, aerosol microphysical properties.

    Parameters
    ----------
    dataset : xarray.Dataset
    var : str
        Variable name.
    iy, ix : int
        Center pixel indices.
    n : int
        Neighborhood half-width.
        n=1 -> 3x3; n=2 -> 5x5.
    wavelength_var : str
        Wavelength coordinate variable.
    xlim, ylim : tuple, optional
        Axis limits.
    title, ylabel : str, optional
        Custom labels.
    figsize : tuple
        Figure size.
    print_stats : bool
        Print center and neighborhood mean.
    ax : matplotlib.axes.Axes, optional
        Existing axes for plotting.

    Returns
    -------
    fig, ax, stats
    """

    data = np.asarray(dataset[var].values)
    wavelength = np.asarray(
        dataset[wavelength_var].values
    ).squeeze()

    if data.ndim != 3:
        raise ValueError(
            f"{var} must have dimensions "
            "(lat, lon, wavelength)"
        )

    ny, nx, nwv = data.shape

    if wavelength.ndim != 1 or len(wavelength) != nwv:
        raise ValueError(
            f"Wavelength shape {wavelength.shape} "
            f"does not match spectral dimension {nwv}"
        )

    if not (0 <= iy < ny and 0 <= ix < nx):
        raise IndexError("Center pixel indices are out of bounds")

    # Spatial neighborhood
    y0 = max(0, iy - n)
    y1 = min(ny, iy + n + 1)
    x0 = max(0, ix - n)
    x1 = min(nx, ix + n + 1)

    box = data[y0:y1, x0:x1, :]

    # Spectral statistics
    center = data[iy, ix, :]
    mean = np.nanmean(box, axis=(0, 1))
    std = np.nanstd(box, axis=(0, 1))

    # Geolocation
    lat = dataset["latitude"].values[iy, ix]
    lon = dataset["longitude"].values[iy, ix]

    # Plot
    if ax is None:
        fig, ax = plt.subplots(figsize=figsize)
    else:
        fig = ax.figure

    ax.fill_between(
        wavelength,
        mean - std,
        mean + std,
        alpha=0.25,
        label=r"$\pm 1$ STD",
    )

    ax.plot(
        wavelength, mean,
        "o-", lw=2,
        label=f"{y1-y0}×{x1-x0} mean",
    )

    ax.plot(
        wavelength, center,
        "o--", lw=1.5,
        label="Center pixel",
    )

    # Automatic labels from variable attributes
    attrs = dataset[var].attrs

    if ylabel is None:
        ylabel = attrs.get(
            "long_name",
            var
        )
        units = attrs.get("units", "")
        if units and units != "1":
            ylabel += f" ({units})"

    if title is None:
        title = (
            f"{var}\n"
            f"({lat:.4f}°, {lon:.4f}°)"
        )

    ax.set_xlabel("Wavelength (nm)")
    ax.set_ylabel(ylabel)
    ax.set_title(title)

    if xlim is not None:
        ax.set_xlim(xlim)

    if ylim is not None:
        ax.set_ylim(ylim)

    ax.grid(alpha=0.3)
    ax.legend()

    if ax is not None:
        fig.tight_layout()

    if print_stats:
        print(f"{var} center:", center)
        print(f"{var} mean:", mean)
        print(f"{var} std:", std)

    stats = {
        "wavelength": wavelength,
        "center": center,
        "mean": mean,
        "std": std,
        "latitude": lat,
        "longitude": lon,
        "bounds": (y0, y1, x0, x1),
        "npixels": (y1-y0) * (x1-x0),
    }

    return fig, ax, stats

########### Plot angular view for a pixel (or within a box)

def plot_rhos_angular(
    dataset, var, iy, ix, n=2,
    wavelengths=(440, 550, 650),
    title=None, ylim=None,
    angle_var="sensor_view_angle",
    wv_var="intensity_wavelength",
):
    """
    Plot SPEXone angular reflectance at selected wavelengths.

    Data dimensions:
        (lat, lon, angle, wavelength)
        Example: (396, 29, 5, 34)

    Wavelength coordinates:
        (wavelength,) or (angle, wavelength)

    For wavelength coordinates that vary by angle, the closest
    spectral channel is selected independently for each angle.
    """

    rhos = np.asarray(dataset[var].values)
    wv = np.asarray(dataset[wv_var].values).squeeze()
    angle = np.asarray(dataset[angle_var].values).squeeze()

    if rhos.ndim != 4:
        raise ValueError(
            "Expected data dimensions (lat, lon, angle, wavelength)"
        )

    ny, nx, nang, nwv = rhos.shape

    # Prepare wavelength coordinates: (angle, wavelength)
    if wv.ndim == 1:
        if wv.size != nwv:
            raise ValueError("Wavelength dimension mismatch")
        wv = np.broadcast_to(wv, (nang, nwv))

    elif wv.ndim == 2:
        if wv.shape != (nang, nwv):
            raise ValueError(
                f"Expected wavelength shape {(nang, nwv)}, "
                f"got {wv.shape}"
            )
    else:
        raise ValueError(
            "Wavelength must be 1D or 2D"
        )

    # Extract center-pixel viewing angles
    if angle.ndim == 3:
        a = angle[iy, ix, :]
    elif angle.ndim == 2:
        if angle.shape == (ny, nang):
            a = angle[iy, :]
        elif angle.shape == (nx, nang):
            a = angle[ix, :]
        else:
            raise ValueError(
                f"Unsupported viewing angle shape: {angle.shape}"
            )
    elif angle.ndim == 1:
        a = angle
    else:
        raise ValueError(
            f"Unsupported viewing angle shape: {angle.shape}"
        )

    if len(a) != nang:
        raise ValueError("Viewing angle dimension mismatch")

    # Spatial box
    y0, y1 = max(0, iy-n), min(ny, iy+n+1)
    x0, x1 = max(0, ix-n), min(nx, ix+n+1)

    box = rhos[y0:y1, x0:x1, :, :]

    # Number of panels
    nplots = len(wavelengths)
    if nplots == 0:
        raise ValueError("At least one wavelength is required")

    ncols = min(2, nplots)
    nrows = math.ceil(nplots / ncols)

    fig, axes = plt.subplots(
        nrows, ncols,
        figsize=(10, 3.5*nrows),
        sharex=True,
        squeeze=False,
    )
    axes = axes.ravel()

    # Sort by viewing angle
    order = np.argsort(a)
    a_sorted = a[order]

    for j, target in enumerate(wavelengths):

        ax = axes[j]

        # Closest wavelength index for each viewing angle
        iw_selected = np.array([
            np.nanargmin(np.abs(wv[ia, :] - target))
            for ia in range(nang)
        ])

        # Actual selected wavelengths
        actual_wv = wv[
            np.arange(nang), iw_selected
        ]

        # Extract data: (lat, lon, angle)
        selected_box = np.stack(
            [
                box[:, :, ia, iw_selected[ia]]
                for ia in range(nang)
            ],
            axis=-1,
        )

        # Spatial statistics
        mean = np.nanmean(selected_box, axis=(0, 1))
        std = np.nanstd(selected_box, axis=(0, 1))

        center = np.array([
            rhos[iy, ix, ia, iw_selected[ia]]
            for ia in range(nang)
        ])

        # Sort by viewing angle
        m = mean[order]
        s = std[order]
        c = center[order]

        ax.fill_between(
            a_sorted, m-s, m+s,
            alpha=0.25,
            label="±1 STD",
        )

        ax.plot(
            a_sorted, m, "o-",
            label=f"{y1-y0}×{x1-x0} mean",
        )

        ax.plot(
            a_sorted, c, "o--",
            label="Center pixel",
        )

        # Display actual wavelength range
        wv_min = np.nanmin(actual_wv)
        wv_max = np.nanmax(actual_wv)

        if np.isclose(wv_min, wv_max):
            wv_label = f"{wv_min:.1f} nm"
        else:
            wv_label = f"{wv_min:.1f}–{wv_max:.1f} nm"

        ax.set_title(
            f"Target {target} nm (actual: {wv_label})"
        )
        ax.set_xlabel("Sensor View Angle (°)")
        ax.set_ylabel(r"$\rho_s$")
        ax.grid(alpha=0.3)

        if ylim is not None:
            ax.set_ylim(ylim)

        # Print wavelength selection for verification
        print(f"\nTarget wavelength: {target} nm")
        for ia in range(nang):
            print(
                f"  Angle {ia}: "
                f"index={iw_selected[ia]}, "
                f"wavelength={actual_wv[ia]:.2f} nm"
            )

    # Remove unused panels
    for ax in axes[nplots:]:
        ax.remove()

    axes[0].legend()

    # Geolocation
    lat = dataset["latitude"].values[iy, ix]
    lon = dataset["longitude"].values[iy, ix]

    fig.suptitle(
        title or
        f"{var}\n({lat:.4f}°, {lon:.4f}°)"
    )

    plt.tight_layout()
    plt.show()

    return fig, axes[:nplots]


def rhos_angle_ratio(
    dataset, var, iy, ix, angle1, angle2,
    angle_var="sensor_view_angle",
    wv_var="intensity_wavelength",
):
    """
    Compute SPEXone reflectance ratios between two viewing angles.

    Data dimensions:
        (lat, lon, angle, wavelength)
        Example: (396, 29, 5, 34)

    Returns
    -------
    pandas.DataFrame
        wavelength, angle1, angle2, rhos1, rhos2, ratio
    """

    rhos = np.asarray(dataset[var].values)
    angle = np.asarray(dataset[angle_var].values).squeeze()
    wv = np.asarray(dataset[wv_var].values).squeeze()

    ny, nx, nang, nwv = rhos.shape

    # Extract viewing angles at center pixel
    if angle.ndim == 3:
        a = angle[iy, ix, :]
    elif angle.ndim == 2:
        if angle.shape == (ny, nang):
            a = angle[iy, :]
        elif angle.shape == (nx, nang):
            a = angle[ix, :]
        else:
            raise ValueError(
                f"Unsupported angle shape: {angle.shape}"
            )
    else:
        a = angle

    # Find nearest viewing angles
    j1 = int(np.nanargmin(np.abs(a - angle1)))
    j2 = int(np.nanargmin(np.abs(a - angle2)))

    # Reflectance spectra at selected angles
    r1 = rhos[iy, ix, j1, :]
    r2 = rhos[iy, ix, j2, :]

    # Extract wavelengths
    if wv.ndim == 1:
        wv1 = wv
        wv2 = wv
    elif wv.ndim == 2:
        wv1 = wv[j1, :]
        wv2 = wv[j2, :]
    else:
        raise ValueError(
            f"Unsupported wavelength shape: {wv.shape}"
        )

    # Wavelength associated with each spectral channel
    wavelength = (wv1 + wv2) / 2

    # Compute ratio, avoiding division by zero
    ratio = np.full(nwv, np.nan, dtype=float)
    np.divide(
        r2, r1,
        out=ratio,
        where=np.isfinite(r1) & (r1 != 0),
    )

    return pd.DataFrame({
        "wavelength": wavelength,
        "angle1": a[j1],
        "angle2": a[j2],
        "rhos1": r1,
        "rhos2": r2,
        "ratio": ratio,
    })

def compute_rhos_angular_mean(
    dataset, var, iy, ix, n=2,
    wavelengths=(440, 550, 650),
    angle_range=(0, 40),
    use_abs_angle=True,
    angle_var="sensor_view_angle",
    wv_var="intensity_wavelength",
):
    """
    Compute SPEXone angular means for selected wavelengths.

    Data dimensions:
        (lat, lon, angle, wavelength)
        Example: (396, 29, 5, 34)

    Wavelength coordinates:
        (wavelength,) or (angle, wavelength)

    Returns
    -------
    pandas.DataFrame
        Angular means for center pixel and spatial neighborhood.
    """

    rhos = np.asarray(dataset[var].values)
    wv = np.asarray(dataset[wv_var].values).squeeze()
    angle = np.asarray(dataset[angle_var].values).squeeze()

    if rhos.ndim != 4:
        raise ValueError(
            "Expected data dimensions (lat, lon, angle, wavelength)"
        )

    ny, nx, nang, nwv = rhos.shape

    # Prepare wavelength coordinates
    if wv.ndim == 1:
        if wv.size != nwv:
            raise ValueError(
                f"Expected {nwv} wavelengths, got {wv.size}"
            )
        wv = np.broadcast_to(wv, (nang, nwv))

    elif wv.ndim == 2:
        if wv.shape != (nang, nwv):
            raise ValueError(
                f"Expected wavelength shape {(nang, nwv)}, "
                f"got {wv.shape}"
            )
    else:
        raise ValueError(
            f"Unsupported wavelength shape: {wv.shape}"
        )

    # Extract viewing angles at center pixel
    if angle.ndim == 3:
        a = angle[iy, ix, :]
    elif angle.ndim == 2:
        if angle.shape == (ny, nang):
            a = angle[iy, :]
        elif angle.shape == (nx, nang):
            a = angle[ix, :]
        else:
            raise ValueError(
                f"Unsupported angle shape: {angle.shape}"
            )
    elif angle.ndim == 1:
        a = angle
    else:
        raise ValueError(
            f"Unsupported angle shape: {angle.shape}"
        )

    if len(a) != nang:
        raise ValueError("Viewing angle dimension mismatch")

    # Spatial neighborhood
    y0 = max(0, iy-n)
    y1 = min(ny, iy+n+1)
    x0 = max(0, ix-n)
    x1 = min(nx, ix+n+1)

    box = rhos[y0:y1, x0:x1, :, :]

    # Safe spatial mean (avoids all-NaN warnings)
    def safe_spatial_mean(arr):
        valid = np.isfinite(arr)
        count = valid.sum(axis=(0, 1))
        total = np.sum(
            np.where(valid, arr, 0.0),
            axis=(0, 1)
        )
        return np.divide(
            total, count,
            out=np.full(count.shape, np.nan, dtype=float),
            where=count > 0
        )

    # Angular range mask
    aa = np.abs(a) if use_abs_angle else a

    mask_range = (
        np.isfinite(aa)
        & (aa >= angle_range[0])
        & (aa <= angle_range[1])
    )

    results = []

    for target in wavelengths:

        # Closest wavelength for each angle
        iw_selected = np.array([
            np.nanargmin(np.abs(wv[ia, :] - target))
            for ia in range(nang)
        ])

        actual_wv = wv[
            np.arange(nang), iw_selected
        ]

        # Extract reflectance for each viewing angle
        selected_box = np.stack([
            box[:, :, ia, iw_selected[ia]]
            for ia in range(nang)
        ], axis=-1)

        # Spatial mean at each angle
        spatial_mean = safe_spatial_mean(selected_box)

        # Center pixel at each angle
        center = np.array([
            rhos[iy, ix, ia, iw_selected[ia]]
            for ia in range(nang)
        ])

        # Valid angle masks
        valid_center = (
            np.isfinite(a)
            & np.isfinite(center)
        )

        valid_spatial = (
            np.isfinite(a)
            & np.isfinite(spatial_mean)
        )

        # Safe angular mean
        def angular_mean(values, mask):
            return (
                np.mean(values[mask])
                if np.any(mask) else np.nan
            )

        results.append({
            "wavelength": target,
            "actual_wavelength": np.nanmean(actual_wv),

            "center_mean_all": angular_mean(
                center, valid_center
            ),
            "center_mean_range": angular_mean(
                center, valid_center & mask_range
            ),

            "spatial_mean_all": angular_mean(
                spatial_mean, valid_spatial
            ),
            "spatial_mean_range": angular_mean(
                spatial_mean, valid_spatial & mask_range
            ),

            "n_angles_center": int(valid_center.sum()),
            "n_angles_range": int(
                (valid_center & mask_range).sum()
            ),
            "n_angles_spatial": int(valid_spatial.sum()),
            "n_angles_spatial_range": int(
                (valid_spatial & mask_range).sum()
            ),
        })

    return pd.DataFrame(results)
```
